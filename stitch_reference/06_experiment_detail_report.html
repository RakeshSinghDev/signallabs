SignalLab SQL/Python Recipe

SELECT 
  dt, 
  close, 
  lag(close, 1) OVER (ORDER BY dt) as prev_close,
  (close - lag(close, 1) OVER (ORDER BY dt)) / lag(close, 1) OVER (ORDER BY dt) as pct_change
FROM nse_indices 
WHERE symbol = 'NIFTY 50' 
  AND pct_change <= -0.05;
